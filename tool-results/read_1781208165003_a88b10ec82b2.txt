   119→   119→  const [rejectReason, setRejectReason] = useState('')

   121→   121→
   122→   122→  /* ---- Tasks state ---- */

   127→   127→
   128→   128→  /* ---- Audit state ---- */

   131→   131→
   132→   132→  /* ---- Notifications state ---- */

   135→   135→
   136→   136→  /* ---- Fetchers ---- */
   137→   137→  const fetchApps = useCallback(async () => {
   138→   138→    const q = wlFilter !== 'ALL' ? `?status=${wlFilter}` : ''
   139→   139→    const res = await fetch(`/api/whitelist${q}`)

   148→   148→  const fetchAudit = useCallback(async () => {
   149→   149→    const res = await fetch(`/api/audit?date=${auditDate}`)

   154→   154→    const res = await fetch(`/api/notifications?userId=${notifUserId}`)

   161→   161→  useEffect(() => { fetchNotifications() }, [fetchNotifications])

   163→   163→  /* ---- Whitelist actions ---- */
   164→   164→  async function handleReview(appId: string, status: 'APPROVED' | 'REJECTED') {

   167→   167→      const body: Record<string, unknown> = { status, reviewerId: 'staff-user' }

   169→   169→      await fetch(`/api/whitelist/${appId}`, {
   170→   170→        method: 'PATCH',
   171→   171→        headers: { 'Content-Type': 'application/json' },
   172→   172→        body: JSON.stringify(body),
   173→   173→      })
   174→   174→      setSelectedApp(null)
   175→   175→      setRejectReason('')
   176→   176→      fetchApps()
   177→   177→    } finally {
   178→   178→      setActionLoading(false)
   179→   179→    }
   180→   180→  }
   181→   181→
   182→   182→  /* ---- Task actions ---- */

   186→   186→      method: 'POST',
   187→   187→      headers: { 'Content-Type': 'application/json' },

   189→   189→    })
   190→   190→    setNewTask({ title: '', priority: 'MEDIUM', notes: '' })
   191→   191→    setShowTaskForm(false)
   192→   192→    fetchTasks()
   193→   193→  }

   197→   197→      method: 'PATCH',
   198→   198→      headers: { 'Content-Type': 'application/json' },
   199→   199→      body: JSON.stringify(data),
   200→   200→    })
   201→   201→    fetchTasks()
   202→   202→  }
   203→   203→
   204→   204→  async function deleteTask(id: string) {
   205→   205→    await fetch(`/api/tasks/${id}`, { method: 'DELETE' })
   206→   206→    fetchTasks()
   207→   207→  }
   208→   208→

   212→   212→      method: 'PATCH',
   213→   213→      headers: { 'Content-Type': 'application/json' },
   214→   214→      body: JSON.stringify(notificationId ? { notificationId } : { userId: notifUserId }),
   215→   215→    })
   216→   216→    fetchNotifications()
   217→   217→  }
   218→   218→
   219→   219→  const unreadCount = notifications.filter((n) => !n.readAt).length
   220→   220→
   221→   221→  /* ---------------------------------------------------------------- */

   230→   230→            value="whitelist"
   231→   231→            className="data-[state=active]:neon-glow data-[state=active]:bg-[rgba(124,58,237,0.2)] data-[state=active]:text-[#a78bfa] flex items-center gap-1.5 text-sm text-gray-400 transition-all"
   232→   232→          >
   233→   233→            <Inbox className="size-4" /> Whitelist Inbox
   234→   234→            {apps.filter((a) => a.status === 'PENDING').length > 0 && (
   235→   235→              <span className="ml-1 rounded-full bg-yellow-500/30 px-1.5 py-0.5 text-[10px] font-bold text-yellow-400">
   236→   236→                {apps.filter((a) => a.status === 'PENDING').length}
   237→   237→              </span>
   238→   238→            )}
   239→   239→          </TabsTrigger>
   240→   240→          <TabsTrigger
   241→   241→            value="tasks"
   242→   242→            className="data-[state=active]:neon-glow data-[state=active]:bg-[rgba(124,58,237,0.2)] data-[state=active]:text-[#a78bfa] flex items-center gap-1.5 text-sm text-gray-400 transition-all"
   243→   243→          >
   244→   244→            <ClipboardList className="size-4" /> Tareas Diarias
   245→   245→          </TabsTrigger>
   246→   246→          <TabsTrigger
   247→   247→            value="audit"
   248→   248→            className="data-[state=active]:neon-glow data-[state=active]:bg-[rgba(124,58,237,0.2)] data-[state=active]:text-[#a78bfa] flex items-center gap-1.5 text-sm text-gray-400 transition-all"
   249→   249→          >
   250→   250→            <Activity className="size-4" /> Actividad del Staff
   251→   251→          </TabsTrigger>
   252→   252→          <TabsTrigger
   253→   253→            value="notifications"
   254→   254→            className="data-[state=active]:neon-glow data-[state=active]:bg-[rgba(124,58,237,0.2)] data-[state=active]:text-[#a78bfa] flex items-center gap-1.5 text-sm text-gray-400 transition-all"
   255→   255→          >
   256→   256→            <Bell className="size-4" /> Notificaciones
   257→   257→            {unreadCount > 0 && (
   258→   258→              <span className="ml-1 rounded-full bg-red-500/30 px-1.5 py-0.5 text-[10px] font-bold text-red-400">
   259→   259→                {unreadCount}
   260→   260→              </span>
   261→   261→            )}
   262→   262→          </TabsTrigger>
   263→   263→        </TabsList>
   264→   264→
   265→   265→        {/* ============================================================ */}
   266→   266→        {/* TAB 1 – Whitelist Inbox                                      */}

   268→   268→        <TabsContent value="whitelist">
   269→   269→          <div className="mb-4 flex flex-wrap items-center gap-3">
   270→   270→            <span className="text-sm text-gray-400">Filtrar:</span>
   271→   271→            {['ALL', 'PENDING', 'APPROVED', 'REJECTED'].map((s) => (
   272→   272→              <Button
   273→   273→                key={s}
   274→   274→                size="sm"
   275→   275→                variant={wlFilter === s ? 'default' : 'outline'}
   276→   276→                className={wlFilter === s ? 'neon-glow' : ''}
   277→   277→                onClick={() => setWlFilter(s)}
   278→   278→              >

   280→   280→              </Button>
   281→   281→            ))}
   282→   282→          </div>
   283→   283→
   284→   284→          <div className="overflow-x-auto rounded-lg border border-[rgba(124,58,237,0.2)]">

   309→   309→                        {app.status}

   313→   313→                    <TableCell>
   314→   314→                      <Button size="sm" variant="ghost" onClick={() => { setSelectedApp(app); setRejectReason('') }}>
   315→   315→                        <Eye className="size-4 mr-1" /> Ver

   317→   317→                    </TableCell>
   318→   318→                  </TableRow>

   320→   320→              </TableBody>

   329→   329→                  <DialogHeader>
   330→   330→                    <DialogTitle className="text-white neon-text-glow">
   331→   331→                      Solicitud de {selectedApp.user.username}

   376→   376→                      </DialogFooter>
   377→   377→                    </div>

   379→   379→

   387→   387→                      )}
   388→   388→                    </div>
