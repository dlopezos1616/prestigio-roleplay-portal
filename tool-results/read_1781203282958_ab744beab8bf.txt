   102→   102→  return new Date(d).toLocaleDateString('es-ES', {

   104→   104→  })
   105→   105→}

   114→   114→export default function StaffPanel() {
   115→   115→  /* ---- Whitelist state ---- */
   116→   116→  const [apps, setApps] = useState<WhitelistApp[]>([])

   118→   118→  const [selectedApp, setSelectedApp] = useState<WhitelistApp | null>(null)
   119→   119→  const [rejectReason, setRejectReason] = useState('')
   120→   120→  const [actionLoading, setActionLoading] = useState(false)
   121→   121→
   122→   122→  /* ---- Tasks state ---- */

   124→   124→  const [taskDate, setTaskDate] = useState(todayStr())

   126→   126→  const [newTask, setNewTask] = useState({ title: '', priority: 'MEDIUM' as const, notes: '' })
   127→   127→
   128→   128→  /* ---- Audit state ---- */
   129→   129→  const [auditLogs, setAuditLogs] = useState<AuditLog[]>([])
   130→   130→  const [auditDate, setAuditDate] = useState(todayStr())
   131→   131→
   132→   132→  /* ---- Notifications state ---- */

   134→   134→  const [notifUserId] = useState('demo-user')
   135→   135→
   136→   136→  /* ---- Fetchers ---- */
   137→   137→  const fetchApps = useCallback(async () => {
   138→   138→    const q = wlFilter !== 'ALL' ? `?status=${wlFilter}` : ''
   139→   139→    const res = await fetch(`/api/whitelist${q}`)
   140→   140→    if (res.ok) setApps(await res.json())
   141→   141→  }, [wlFilter])
   142→   142→
   143→   143→  const fetchTasks = useCallback(async () => {
   144→   144→    const res = await fetch(`/api/tasks?date=${taskDate}`)
   145→   145→    if (res.ok) setTasks(await res.json())
   146→   146→  }, [taskDate])
   147→   147→
   148→   148→  const fetchAudit = useCallback(async () => {
   149→   149→    const res = await fetch(`/api/audit?date=${auditDate}`)
   150→   150→    if (res.ok) setAuditLogs(await res.json())
   151→   151→  }, [auditDate])
   152→   152→
   153→   153→  const fetchNotifications = useCallback(async () => {
   154→   154→    const res = await fetch(`/api/notifications?userId=${notifUserId}`)
   155→   155→    if (res.ok) setNotifications(await res.json())
   156→   156→  }, [notifUserId])
   157→   157→
   158→   158→  useEffect(() => { fetchApps() }, [fetchApps])
   159→   159→  useEffect(() => { fetchTasks() }, [fetchTasks])
   160→   160→  useEffect(() => { fetchAudit() }, [fetchAudit])
   161→   161→  useEffect(() => { fetchNotifications() }, [fetchNotifications])
   162→   162→
   163→   163→  /* ---- Whitelist actions ---- */
   164→   164→  async function handleReview(appId: string, status: 'APPROVED' | 'REJECTED') {
   165→   165→    setActionLoading(true)
   166→   166→    try {
   167→   167→      const body: Record<string, unknown> = { status, reviewerId: 'staff-user' }
   168→   168→      if (status === 'REJECTED' && rejectReason) body.rejectReason = rejectReason
   169→   169→      await fetch(`/api/whitelist/${appId}`, {
   170→   170→        method: 'PATCH',
   171→   171→        headers: { 'Content-Type': 'application/json' },
   172→   172→        body: JSON.stringify(body),
   173→   173→      })
   174→   174→      setSelectedApp(null)
   175→   175→      setRejectReason('')
   176→   176→      fetchApps()
   177→   177→    } finally {
   178→   178→      setActionLoading(false)
   179→   179→    }
   180→   180→  }
   181→   181→
   182→   182→  /* ---- Task actions ---- */
   183→   183→  async function addTask() {
   184→   184→    if (!newTask.title.trim()) return
   185→   185→    await fetch('/api/tasks', {
   186→   186→      method: 'POST',
   187→   187→      headers: { 'Content-Type': 'application/json' },
   188→   188→      body: JSON.stringify({ ...newTask, date: taskDate }),
   189→   189→    })
   190→   190→    setNewTask({ title: '', priority: 'MEDIUM', notes: '' })
   191→   191→    setShowTaskForm(false)
   192→   192→    fetchTasks()
   193→   193→  }
   194→   194→
   195→   195→  async function updateTask(id: string, data: Partial<Task>) {
   196→   196→    await fetch(`/api/tasks/${id}`, {
   197→   197→      method: 'PATCH',
   198→   198→      headers: { 'Content-Type': 'application/json' },
   199→   199→      body: JSON.stringify(data),
   200→   200→    })
   201→   201→    fetchTasks()
   202→   202→  }
   203→   203→
   204→   204→  async function deleteTask(id: string) {
   205→   205→    await fetch(`/api/tasks/${id}`, { method: 'DELETE' })
   206→   206→    fetchTasks()
   207→   207→  }
   208→   208→
   209→   209→  /* ---- Notification actions ---- */

   212→   212→      method: 'PATCH',
   213→   213→      headers: { 'Content-Type': 'application/json' },
   214→   214→      body: JSON.stringify(notificationId ? { notificationId } : { userId: notifUserId }),
   215→   215→    })
   216→   216→    fetchNotifications()
   217→   217→  }
   218→   218→
   219→   219→  const unreadCount = notifications.filter((n) => !n.readAt).length
   220→   220→
   221→   221→  /* ---------------------------------------------------------------- */
   222→   222→  /*  Render                                                           */
   223→   223→  /* ---------------------------------------------------------------- */
   224→   224→  return (

   226→   226→      <h2 className="mb-4 text-xl font-bold text-white neon-text-glow">
   227→   227→        Panel de Staff
   228→   228→      </h2>
   229→   229→
   230→   230→      <Tabs defaultValue="whitelist" className="w-full">
   231→   231→        {/* ---- Tab navigation ---- */}

   233→   233→          <TabsTrigger
   234→   234→            value="whitelist"
   235→   235→            className="data-[state=active]:neon-glow data-[state=active]:bg-[rgba(124,58,237,0.2)] data-[state=active]:text-[#a78bfa] flex items-center gap-1.5 text-sm text-gray-400 transition-all"
   236→   236→          >
   237→   237→            <Inbox className="size-4" /> Whitelist Inbox
   238→   238→            {apps.filter((a) => a.status === 'PENDING').length > 0 && (
   239→   239→              <span className="ml-1 rounded-full bg-yellow-500/30 px-1.5 py-0.5 text-[10px] font-bold text-yellow-400">
   240→   240→                {apps.filter((a) => a.status === 'PENDING').length}
   241→   241→              </span>
   242→   242→            )}
   243→   243→          </TabsTrigger>
   244→   244→          <TabsTrigger
   245→   245→            value="tasks"
   246→   246→            className="data-[state=active]:neon-glow data-[state=active]:bg-[rgba(124,58,237,0.2)] data-[state=active]:text-[#a78bfa] flex items-center gap-1.5 text-sm text-gray-400 transition-all"
   247→   247→          >
   248→   248→            <ClipboardList className="size-4" /> Tareas Diarias
   249→   249→          </TabsTrigger>
   250→   250→          <TabsTrigger
   251→   251→            value="audit"
   252→   252→            className="data-[state=active]:neon-glow data-[state=active]:bg-[rgba(124,58,237,0.2)] data-[state=active]:text-[#a78bfa] flex items-center gap-1.5 text-sm text-gray-400 transition-all"
   253→   253→          >
   254→   254→            <Activity className="size-4" /> Actividad del Staff
   255→   255→          </TabsTrigger>
   256→   256→          <TabsTrigger
   257→   257→            value="notifications"
   258→   258→            className="data-[state=active]:neon-glow data-[state=active]:bg-[rgba(124,58,237,0.2)] data-[state=active]:text-[#a78bfa] flex items-center gap-1.5 text-sm text-gray-400 transition-all"
   259→   259→          >
   260→   260→            <Bell className="size-4" /> Notificaciones
   261→   261→            {unreadCount > 0 && (
   262→   262→              <span className="ml-1 rounded-full bg-red-500/30 px-1.5 py-0.5 text-[10px] font-bold text-red-400">
   263→   263→                {unreadCount}
   264→   264→              </span>
   265→   265→            )}
   266→   266→          </TabsTrigger>
   267→   267→        </TabsList>
   268→   268→
   269→   269→        {/* ============================================================ */}
   270→   270→        {/* TAB 1 – Whitelist Inbox                                      */}
   271→   271→        {/* ============================================================ */}
   272→   272→        <TabsContent value="whitelist">
   273→   273→          <div className="mb-4 flex flex-wrap items-center gap-3">
   274→   274→            <span className="text-sm text-gray-400">Filtrar:</span>
   275→   275→            {['ALL', 'PENDING', 'APPROVED', 'REJECTED'].map((s) => (
   276→   276→              <Button
   277→   277→                key={s}
   278→   278→                size="sm"
   279→   279→                variant={wlFilter === s ? 'default' : 'outline'}
   280→   280→                className={wlFilter === s ? 'neon-glow' : ''}
   281→   281→                onClick={() => setWlFilter(s)}
   282→   282→              >
   283→   283→                {s === 'ALL' ? 'Todos' : s === 'PENDING' ? 'Pendientes' : s === 'APPROVED' ? 'Aprobados' : 'Rechazados'}
   284→   284→              </Button>
   285→   285→            ))}
   286→   286→          </div>
   287→   287→
   288→   288→          <div className="overflow-x-auto rounded-lg border border-[rgba(124,58,237,0.2)]">
   289→   289→            <Table>
   290→   290→              <TableHeader>
   291→   291→                <TableRow className="border-[rgba(124,58,237,0.2)] hover:bg-transparent">
   292→   292→                  <TableHead className="text-gray-400">Usuario</TableHead>
   293→   293→                  <TableHead className="text-gray-400">Intento</TableHead>
   294→   294→                  <TableHead className="text-gray-400">Estado</TableHead>

   302→   302→                    <TableCell colSpan={5} className="text-center text-gray-500 py-8">
   303→   303→                      No hay solicitudes de whitelist

   305→   305→                  </TableRow>
   306→   306→                )}
   307→   307→                {apps.map((app) => (
   308→   308→                  <TableRow key={app.id} className="border-[rgba(124,58,237,0.15)] hover:bg-[rgba(124,58,237,0.05)]">
   309→   309→                    <TableCell className="font-medium text-white">{app.user.username}</TableCell>
   310→   310→                    <TableCell className="text-gray-400">#{app.attemptNumber}</TableCell>
   311→   311→                    <TableCell>
   312→   312→                      <Badge className={`${statusColor[app.status]} border text-xs`}>
   313→   313→                        {app.status}
   314→   314→                      </Badge>
   315→   315→                    </TableCell>
   316→   316→                    <TableCell className="text-gray-400 text-xs">{fmtDate(app.createdAt)}</TableCell>
   317→   317→                    <TableCell>
   318→   318→                      <Button size="sm" variant="ghost" onClick={() => { setSelectedApp(app); setRejectReason('') }}>
   319→   319→                        <Eye className="size-4 mr-1" /> Ver
   320→   320→                      </Button>
   321→   321→                    </TableCell>
   322→   322→                  </TableRow>
   323→   323→                ))}
   324→   324→              </TableBody>
   325→   325→            </Table>
   326→   326→          </div>
   327→   327→
   328→   328→          {/* Application detail dialog */}
   329→   329→          <Dialog open={!!selectedApp} onOpenChange={(o) => { if (!o) setSelectedApp(null) }}>

   332→   332→                <>
   333→   333→                  <DialogHeader>
   334→   334→                    <DialogTitle className="text-white neon-text-glow">
   335→   335→                      Solicitud de {selectedApp.user.username}
   336→   336→                    </DialogTitle>
   337→   337→                    <DialogDescription className="text-gray-400">

   342→   342→                  <div className="space-y-4 py-2">

   353→   353→                      </div>

   355→   355→                  </div>
   356→   356→

   358→   358→                    <div className="space-y-3 pt-2">
   359→   359→                      <Textarea
   360→   360→                        placeholder="Razón de rechazo (obligatorio si rechazas)..."

   371→   371→                          <XCircle className="size-4 mr-1" /> Rechazar

   375→   375→                          onClick={() => handleReview(selectedApp.id, 'APPROVED')}
   376→   376→                          className="bg-green-600 hover:bg-green-700 text-white"
   377→   377→                        >
   378→   378→                          <CheckCircle2 className="size-4 mr-1" /> Aprobar
   379→   379→                        </Button>
   380→   380→                      </DialogFooter>
   381→   381→                    </div>
   382→   382→                  )}
   383→   383→
   384→   384→                  {selectedApp.status !== 'PENDING' && (
   385→   385→                    <div className="flex items-center gap-2 pt-2">
   386→   386→                      <Badge className={`${statusColor[selectedApp.status]} border`}>
   387→   387→                        {selectedApp.status}
   388→   388→                      </Badge>
   389→   389→                      {selectedApp.rejectReason && (

   391→   391→                      )}
   392→   392→                    </div>

   394→   394→                </>
   395→   395→              )}
   396→   396→            </DialogContent>
   397→   397→          </Dialog>
   398→   398→        </TabsContent>
   399→   399→
   400→   400→        {/* ============================================================ */}
   401→   401→        {/* TAB 2 – Tareas Diarias                                       */}
   402→   402→        {/* ============================================================ */}
   403→   403→        <TabsContent value="tasks">

   424→   424→              <div className="flex flex-wrap gap-3">
   425→   425→                <Select value={newTask.priority} onValueChange={(v) => setNewTask({ ...newTask, priority: v as Task['priority'] })}>
