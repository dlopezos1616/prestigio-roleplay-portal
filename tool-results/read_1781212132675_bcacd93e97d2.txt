   318→function LocationMarker({
   319→  location,
   320→  isSelected,
   321→  isHovered,
   322→  onSelect,
   323→  onHover,
   324→  visible,
   325→}: {
   326→  location: MapLocation
   327→  isSelected: boolean
   328→  isHovered: boolean
   329→  onSelect: () => void
   330→  onHover: (id: string | null) => void
   331→  visible: boolean
   332→}) {
   333→  const IconComponent = location.icon
   334→
   335→  return (
   336→    <AnimatePresence>
   337→      {visible && (
   338→        <motion.g
   339→          initial={{ opacity: 0, scale: 0 }}
   340→          animate={{ opacity: 1, scale: 1 }}
   341→          exit={{ opacity: 0, scale: 0 }}
   342→          transition={{ duration: 0.3, type: 'spring', stiffness: 200 }}
   343→          style={{ cursor: 'pointer' }}
   344→          onClick={onSelect}
   345→          onMouseEnter={() => onHover(location.id)}
   346→          onMouseLeave={() => onHover(null)}
   347→        >
   348→          {/* Pulse ring - outer */}
   349→          <motion.circle
   350→            cx={`${location.x}%`}
   351→            cy={`${location.y}%`}
   352→            r={isSelected ? 14 : isHovered ? 12 : 8}
   353→            fill="none"
   354→            stroke={location.color}
   355→            strokeWidth={isSelected ? 2 : 1}
   356→            opacity={isSelected ? 0.6 : isHovered ? 0.4 : 0.2}
   357→            animate={
   358→              isSelected
   359→                ? { r: [10, 16, 10], opacity: [0.6, 0.2, 0.6] }
   360→                : isHovered
   361→                ? { r: [8, 13, 8], opacity: [0.4, 0.15, 0.4] }
   362→                : {}
   363→            }
   364→            transition={{ duration: 1.5, repeat: Infinity, ease: 'easeInOut' }}
   365→          />
   366→
   367→          {/* Second pulse ring for selected */}
   368→          {isSelected && (
   369→            <motion.circle
   370→              cx={`${location.x}%`}
   371→              cy={`${location.y}%`}
   372→              r={18}
   373→              fill="none"
   374→              stroke={location.color}
   375→              strokeWidth={1}
   376→              animate={{ r: [14, 22, 14], opacity: [0.3, 0, 0.3] }}
   377→              transition={{ duration: 2, repeat: Infinity, ease: 'easeInOut', delay: 0.3 }}
   378→            />
   379→          )}
   380→
   381→          {/* Glow circle */}
   382→          <circle
   383→            cx={`${location.x}%`}
   384→            cy={`${location.y}%`}
   385→            r={isHovered || isSelected ? 7 : 5}
   386→            fill={location.color}
   387→            opacity={0.15}
   388→            filter="url(#marker-glow)"
   389→          />
   390→
   391→          {/* Main dot */}
   392→          <motion.circle
   393→            cx={`${location.x}%`}
   394→            cy={`${location.y}%`}
   395→            r={isSelected ? 6 : isHovered ? 5.5 : 4}
   396→            fill={location.color}
   397→            style={{
   398→              filter: `drop-shadow(0 0 4px ${location.color}) drop-shadow(0 0 8px ${location.color}40)`,
   399→            }}
   400→            animate={
   401→              isSelected
   402→                ? { scale: [1, 1.15, 1] }
   403→                : isHovered
   404→                ? { scale: [1, 1.1, 1] }
   405→                : {}
   406→            }
   407→            transition={{ duration: 0.8, repeat: Infinity, ease: 'easeInOut' }}
   408→          />
   409→
   410→          {/* Inner white dot */}
   411→          <circle
   412→            cx={`${location.x}%`}
   413→            cy={`${location.y}%`}
   414→            r={1.5}
   415→            fill="white"
   416→            opacity={0.8}
   417→          />
   418→
   419→          {/* Icon at marker - only on hover/select */}
   420→          {(isHovered || isSelected) && (
   421→            <motion.foreignObject
   422→              x={`${location.x - 1.5}%`}
   423→              y={`${location.y - 3}%`}
   424→              width="3%"
   425→              height="3%"
   426→              initial={{ opacity: 0, scale: 0.5 }}
   427→              animate={{ opacity: 1, scale: 1 }}
   428→              exit={{ opacity: 0, scale: 0.5 }}
   429→              style={{ overflow: 'visible' }}
   430→            >
   431→              <div className="flex items-center justify-center w-full h-full">
   432→                <IconComponent
   433→                  style={{ color: location.color, width: '100%', height: '100%', maxWidth: 16, maxHeight: 16, filter: `drop-shadow(0 0 3px ${location.color})` }}
   434→                />
   435→              </div>
   436→            </motion.foreignObject>
   437→          )}
   438→
   439→          {/* Label tooltip on hover */}
   440→          {(isHovered || isSelected) && (
   441→            <motion.g
   442→              initial={{ opacity: 0, y: 5 }}
   443→              animate={{ opacity: 1, y: 0 }}
   444→              exit={{ opacity: 0, y: 5 }}
   445→              transition={{ duration: 0.2 }}
   446→            >
   447→              <rect
   448→                x={`${location.x - 8}%`}
   449→                y={`${location.y - 10}%`}
   450→                width="16%"
   451→                height="5%"
   452→                rx="3"
   453→                fill="#0f172a"
   454→                stroke={location.color}
   455→                strokeWidth="0.5"
   456→                opacity="0.95"
   457→              />
   458→              <text
   459→                x={`${location.x}%`}
   460→                y={`${location.y - 7.2}%`}
   461→                textAnchor="middle"
   462→                fill={location.color}
   463→                fontSize="3.2"
   464→                fontFamily="system-ui, sans-serif"
   465→                fontWeight="600"
   466→                style={{ textShadow: `0 0 6px ${location.color}60` }}
   467→              >
   468→                {location.name}
   469→              </text>
   470→            </motion.g>
   471→          )}
   472→        </motion.g>
   473→      )}
   474→    </AnimatePresence>
   475→  )
   476→}
